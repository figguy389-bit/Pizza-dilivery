        let cart = {};

        function setLanguage(lang) {
            document.getElementById('lang-modal').style.display = 'none';
        }

        function renderMenu(filter = 'all') {
            const container = document.getElementById('menu-container');
            container.innerHTML = '';
            
            let filtered = filter === 'all' ? menuData : menuData.filter(item => item.cat === filter);
            
            let grid = document.createElement('div');
            grid.className = 'menu-grid';
            
            filtered.forEach(item => {
                let card = document.createElement('div');
                card.className = 'menu-item';
                card.innerHTML = `
                    <img src="${item.img}" alt="${item.name}" onerror="this.src='image_1.png'">
                    <div class="item-details">
                        <h3>${item.name}</h3>
                        <div class="price">${item.price}</div>
                        <button class="add-to-cart" onclick="addToCart(${item.id})">إضافة للطلب 🛒</button>
                    </div>
                `;
                grid.appendChild(card);
            });
            container.appendChild(grid);
        }

        function filterCategory(cat) {
            document.querySelectorAll('.cat-btn').forEach(btn => btn.classList.remove('active'));
            event.target.classList.add('active');
            renderMenu(cat);
        }

        function addToCart(id) {
            const item = menuData.find(i => i.id === id);
            if (cart[id]) {
                cart[id].qty++;
            } else {
                cart[id] = { ...item, qty: 1 };
            }
            updateCartCount();
        }

        function updateCartCount() {
            let total = Object.values(cart).reduce((sum, i) => sum + i.qty, 0);
            document.getElementById('cart-count').innerText = total;
        }

        function openCartModal() {
            const list = document.getElementById('cart-items-list');
            list.innerHTML = '';
            let items = Object.values(cart);
            if (items.length === 0) {
                list.innerHTML = 'السلة فارغة حالياً';
            } else {
                items.forEach(i => {
                    list.innerHTML += `<div>${i.name} (${i.qty}x) - ${i.price}</div>`;
                });
            }
            document.getElementById('cart-modal').style.display = 'flex';
        }

        function closeCartModal() {
            document.getElementById('cart-modal').style.display = 'none';
        }

        function sendOrderWhatsApp(phone) {
            let items = Object.values(cart);
            if (items.length === 0) {
                alert('الرجاء اختيار بعض الأطباق أولاً!');
                return;
            }
            let msg = "مرحباً، أود طلب الأطباق التالية من بيتزا دليفري:\n\n";
            items.forEach(item => {
                msg += `- ${item.name} (${item.qty}x) - ${item.price}\n`;
            });
            msg += `\nعنوان التوصيل (زاخو): `;
            
            let url = `https://wa.me/${phone}?text=${encodeURIComponent(msg)}`;
            window.open(url, '_blank');
        }

        // لوحة التحكم الخاصة
        function openAdminLogin() {
            document.getElementById('admin-modal').style.display = 'flex';
        }
        function closeAdminLogin() {
            document.getElementById('admin-modal').style.display = 'none';
        }
        function verifyAdmin() {
            let code = document.getElementById('admin-code').value;
            if (code === '30067') {
                alert('مرحباً بك في لوحة تحكم وتعديل القائمة الخاصة بك!');
                closeAdminLogin();
            } else {
                alert('كود الدخول غير صحيح!');
            }
        }
