        let menuData = JSON.parse(localStorage.getItem('restaurant_menu')) || defaultMenu;
        let cart = [];

        function setLanguage(lang) {
            document.getElementById('lang-screen').style.display = 'none';
        }

        function renderMenu() {
            const container = document.getElementById('menu-container');
            container.innerHTML = '';
            menuData.forEach((cat, catIdx) => {
                let catDiv = document.createElement('div');
                catDiv.innerHTML = `<div class="category-title">${cat.category}</div>`;
                let gridDiv = document.createElement('div');
                gridDiv.className = 'items-grid';

                cat.items.forEach((item, itemIdx) => {
                    let card = document.createElement('div');
                    card.className = 'item-card';
                    card.innerHTML = `
                        <div>
                            <img src="${item.img}" alt="${item.name}">
                            <div class="item-info">
                                <h3>${item.name}</h3>
                                <div class="price">${item.price}</div>
                            </div>
                        </div>
                        <button onclick="addToCart('${item.name}', '${item.price}')">إضافة للسلة</button>
                    `;
                    gridDiv.appendChild(card);
                });
                catDiv.appendChild(gridDiv);
                container.appendChild(catDiv);
            });
        }

        function addToCart(name, price) {
            cart.push({ name, price });
            updateCart();
        }

        function updateCart() {
            document.getElementById('cart-count').innerText = cart.length;
            let total = cart.reduce((sum, item) => sum + parseInt(item.price.replace(/[^0-9]/g, '') || 0), 0);
            document.getElementById('cart-total').innerText = total;
        }

        function checkout() {
            if (cart.length === 0) {
                alert('السلة فارغة!');
                return;
            }
            let text = "مرحباً، أريـد طلب الأطباق التالية:%0A";
            cart.forEach((i, index) => {
                text += `${index + 1}. ${i.name} - ${i.price}%0A`;
            });
            let total = document.getElementById('cart-total').innerText;
            text += `%0Aالمجموع الكلي: ${total} دينار`;
            
            // إرسال للواتساب (رقم 1 أو 2)
            window.open(`https://wa.me/9647508653006?text=${text}`, '_blank');
        }

        function openAdmin() {
            let pass = prompt("أدخل كود الدخول لقائمة التعديل:");
            if (pass === "30067") {
                let newJson = prompt("عدل بيانات القائمة (JSON):", JSON.stringify(menuData, null, 2));
                if (newJson) {
                    try {
                        menuData = JSON.parse(newJson);
                        localStorage.setItem('restaurant_menu', JSON.stringify(menuData));
                        renderMenu();
                        alert("تم حفظ التعديلات بنجاح وستظهر لدى الجميع!");
                    } catch (e) {
                        alert("خطأ في تنسيق البيانات!");
                    }
                }
            } else if (pass !== null) {
                alert("الكود غير صحيح!");
            }
        }

        renderMenu();
    </script>
