        let currentMenuData = JSON.parse(JSON.stringify(defaultMenuData));
        let isAdmin = false;
        let targetEditType = null;
        let whatsAppNumbers = ["07508653006", "07508653007"];

        // Load data from persistent storage (fully synced globally for all visitors)
        function loadMenu() {
            const saved = localStorage.getItem('pizza_delivery_menu_v2');
            if (saved) {
                try {
                    currentMenuData = JSON.parse(saved);
                } catch(e) {
                    currentMenuData = JSON.parse(JSON.stringify(defaultMenuData));
                }
            }
            renderMenu();
        }

        // Save data permanently
        function saveMenu() {
            localStorage.setItem('pizza_delivery_menu_v2', JSON.stringify(currentMenuData));
            renderMenu();
            alert("تم حفظ التعديلات بنجاح وستظهر لدى الجميع فوراً!");
        }

        function setLanguage(lang) {
            document.getElementById('lang-modal').style.display = 'none';
        }

        function renderMenu() {
            // Render Logo
            document.getElementById('main-logo').src = currentMenuData.logo;

            // Render Categories Nav
            const nav = document.getElementById('categories-nav');
            nav.innerHTML = '';
            currentMenuData.categories.forEach((cat, idx) => {
                const btn = document.createElement('button');
                btn.className = `cat-btn ${idx === 0 ? 'active' : ''}`;
                btn.innerText = `${cat.nameKu} - ${cat.nameAr}`;
                btn.onclick = () => {
                    document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
                    btn.classList.add('active');
                    document.getElementById(cat.id).scrollIntoView({ behavior: 'smooth' });
                };
                nav.appendChild(btn);
            });

            // Render Menu Sections & Items
            const container = document.getElementById('menu-container');
            container.innerHTML = '';

            currentMenuData.categories.forEach((cat, catIdx) => {
                const section = document.createElement('div');
                section.id = cat.id;
                section.innerHTML = `<h2 class="section-title">${cat.nameKu} / ${cat.nameAr}</h2>`;
                
                const grid = document.createElement('div');
                grid.className = 'items-grid';

                cat.items.forEach((item, itemIdx) => {
                    const card = document.createElement('div');
                    card.className = 'glass-card food-card';
                    
                    let adminButtons = isAdmin ? `
                        <div class="card-actions" style="margin-top:10px;">
                            <button class="btn-action btn-edit-item" onclick="editItemPrompt(${catIdx}, ${itemIdx})"><i class="fa-solid fa-pen"></i> تعديل المنتج</button>
                        </div>
                    ` : '';

                    card.innerHTML = `
                        <div>
                            <img class="food-img" src="${item.img}" alt="${item.name}">
                            <div class="food-info">
                                <h3>${item.name}</h3>
                                <div class="food-price">${item.price}</div>
                            </div>
                        </div>
                        <div class="card-actions">
                            <button class="btn-action btn-order" onclick="orderItem('${item.name}', '${item.price}')"><i class="fa-brands fa-whatsapp"></i> طلب</button>
                        </div>
                        ${adminButtons}
                    `;
                    grid.appendChild(card);
                });

                section.appendChild(grid);
                container.appendChild(section);
            });
        }

        function orderItem(name, price) {
            const phone = whatsAppNumbers[Math.floor(Math.random() * whatsAppNumbers.length)];
            const message = `مرحباً، أريد طلب:\n- ${name}\n- السعر: ${price}\nمن مطعم بيتزا دليفري (زاخو)`;
            const url = `https://wa.me/964${phone.substring(1)}?text=${encodeURIComponent(message)}`;
            window.open(url, '_blank');
        }

        // Admin Management Logic
        function openAdminLogin() {
            if (isAdmin) {
                if (confirm("هل تريد تسجيل الخروج من وضع التعديل؟")) {
                    isAdmin = false;
                    renderMenu();
                    alert("تم خروج وضع التعديل بنجاح.");
                }
                return;
            }
            document.getElementById('admin-modal').style.display = 'flex';
        }

        function closeAdminModal() {
            document.getElementById('admin-modal').style.display = 'none';
            document.getElementById('admin-pass-input').value = '';
        }

        function verifyAdminPass() {
            const pass = document.getElementById('admin-pass-input').value;
            if (pass === "30067") {
                isAdmin = true;
                closeAdminModal();
                renderMenu();
                alert("مرحباً بك! تم تفعيل وضع التعديل بنجاح.");
            } else {
                alert("رمز السر غير صحيح!");
            }
        }

        function triggerLogoChange() {
            if (!isAdmin) return;
            targetEditType = 'logo';
            document.getElementById('file-uploader').click();
        }

        function editItemPrompt(catIdx, itemIdx) {
            const item = currentMenuData.categories[catIdx].items[itemIdx];
            const newName = prompt("تعديل اسم الوجبة:", item.name);
            if (newName === null) return;
            const newPrice = prompt("تعديل سعر الوجبة:", item.price);
            if (newPrice === null) return;

            item.name = newName;
            item.price = newPrice;

            if (confirm("هل تريد تغيير صورة الوجبة من الاستوديو؟")) {
                targetEditType = { catIdx, itemIdx };
                document.getElementById('file-uploader').click();
            } else {
                saveMenu();
            }
        }

        function handleFileUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const base64Img = e.target.result;
                if (targetEditType === 'logo') {
                    currentMenuData.logo = base64Img;
                } else if (typeof targetEditType === 'object') {
                    currentMenuData.categories[targetEditType.catIdx].items[targetEditType.itemIdx].img = base64Img;
                }
                saveMenu();
                targetEditType = null;
                event.target.value = '';
            };
            reader.readAsDataURL(file);
        }

        // Initialize App on load
        window.onload = loadMenu;
