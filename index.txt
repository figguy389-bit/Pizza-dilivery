        function setLanguage(lang) {
            document.getElementById('welcomeScreen').style.display = 'none';
            document.getElementById('mainMenu').style.display = 'block';
        }

        function requestAdminAccess() {
            let pass = prompt("ادخل رمز السر");
            if (pass === "30067") {
                document.getElementById('adminPanel').style.display = 'block';
                alert("تم تسجيل الدخول بنجاح إلى لوحة التعديل!");
            } else if (pass !== null) {
                alert("رمز السر خطأ!");
            }
        }

        function changeLogo(event) {
            let reader = new FileReader();
            reader.onload = function(){
                let output = document.getElementById('displayLogo');
                output.src = reader.result;
                localStorage.setItem('restaurant_logo', reader.result);
            };
            reader.readAsDataURL(event.target.files[0]);
        }

        function saveChanges() {
            alert("تم حفظ التعديلات بنجاح! لكي تظهر لدى جميع الزوار بشكل دائم، تأكد من ربط الموقع بقاعدة بيانات سحابية (مثل Firebase).");
        }

        window.onload = function() {
            let savedLogo = localStorage.getItem('restaurant_logo');
            if (savedLogo) {
                document.getElementById('displayLogo').src = savedLogo;
            }
        }
